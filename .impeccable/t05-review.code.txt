async (page) => {
  const metadata = {"purchase": {"packageId": "1d8d9831-aaaf-4dbb-a6fb-9c90f19e7298", "learnerId": "023abbc7-80a2-43f1-9319-28458dc7d3c5", "policyId": "2b278d56-2de5-4cde-9172-00315dbd2d35", "offerId": "bc7fd8af-fcac-4233-86db-e965f4c31144"}, "coverage": {"packageId": "4dbf919a-9618-486d-b902-7c382dcf0266", "sourceId": "28073cc2-440d-4d12-a86d-40aa705efdc5", "enrollmentId": "ae61e1e4-3ea7-4a45-9e44-0560beaa356a"}, "value": {"packageId": "3421a6a3-32d2-4c7f-a67a-766a345e89c3", "enrollmentId": "95bd2a55-1ac3-4956-97ec-ad25525f217b", "proposalId": "ca865b91-527e-455a-ad53-286f28f618fa"}, "conversion": {"packageId": "092376eb-2d24-49d2-a9f9-bc1244a1c391", "targetId": "b1745ca7-ee4a-4604-9ab3-e119cf188e26", "proposalId": "c452a0d9-e8c9-4069-9bb6-174a2bb7cc13", "learnerId": "023abbc7-80a2-43f1-9319-28458dc7d3c5"}};
  const findings = [];
  let consoleErrors = [], warnings = [], failedResponses = [];
  page.on('console', message => { if (message.type() === 'error') consoleErrors.push(message.text().replace(/https?:\/\/\S+/g, '[url]')); if (message.type() === 'warning') warnings.push(message.text().replace(/https?:\/\/\S+/g, '[url]')); });
  page.on('pageerror', error => consoleErrors.push(error.message.replace(/https?:\/\/\S+/g, '[url]')));
  page.on('response', response => { if (response.status() >= 400) failedResponses.push({ path: new URL(response.url()).pathname, status: response.status() }); });
  async function capture(label, target, rtl = false) {
    const measurements = [];
    for (const viewport of [{ label: 'desktop', width: 1440, height: 1000 }, { label: 'mobile', width: 360, height: 780 }]) {
      await page.evaluate(() => document.activeElement?.blur());
      await page.setViewportSize({ width: viewport.width, height: viewport.height });
      await page.evaluate(() => document.fonts.ready);
      if (target) await target.evaluate(element => element.scrollIntoView({ block: 'start', behavior: 'instant' }));
      else await page.evaluate(() => scrollTo({ top: 0, behavior: 'instant' }));
      await page.screenshot({ path: '.impeccable/review/t05-' + label + '-' + viewport.label + '.png', caret: 'initial' });
      measurements.push({ viewport: viewport.width, overflow: await page.evaluate(() => document.documentElement.scrollWidth > innerWidth) });
    }
    await page.evaluate(() => { document.activeElement?.blur(); scrollTo({ top: 0, behavior: 'instant' }); });
    await page.keyboard.press('Tab');
    const focus = await page.evaluate(() => { const element = document.activeElement; const style = getComputedStyle(element); return { tag: element.tagName, text: element.textContent.trim().slice(0, 80), outlineStyle: style.outlineStyle, outlineWidth: style.outlineWidth }; });
    if (rtl) {
      await page.evaluate(() => { document.documentElement.dir = 'rtl'; });
      measurements.push({ viewport: 360, direction: 'rtl', overflow: await page.evaluate(() => document.documentElement.scrollWidth > innerWidth) });
      if (target) await target.evaluate(element => element.scrollIntoView({ block: 'start', behavior: 'instant' }));
      await page.screenshot({ path: '.impeccable/review/t05-' + label + '-rtl.png', caret: 'initial' });
      await page.evaluate(() => { document.documentElement.dir = 'ltr'; });
    }
    findings.push({ surface: label, measurements, focus, consoleErrors, warnings, failedResponses });
    consoleErrors = []; warnings = []; failedResponses = [];
  }
  await page.context().setStorageState('.local/ui-review/t05-desktop-customer.json');
  await page.goto('http://127.0.0.1:3000/learn/packages/' + metadata.purchase.packageId);
  await page.getByRole('heading', { name: 'Purchased credits are waiting for payment', exact: true }).waitFor();
  await capture('purchase-charge', page.getByRole('region', { name: 'Package purchase', exact: true }), true);
  await capture('usable-credits', page.getByRole('heading', { name: 'Usable teaching credits', exact: true }), true);
  await page.getByRole('heading', { name: 'Credits issued', exact: true }).first().waitFor();
  await capture('customer-history', page.locator('details').filter({ has: page.locator('summary').filter({ hasText: /^Credit history$/ }) }), true);
  await page.context().setStorageState('.local/ui-review/t05-desktop-coverage-customer.json');
  await page.goto('http://127.0.0.1:3000/learn/bookings/' + metadata.coverage.enrollmentId);
  await page.locator('#package-coverage').getByRole('heading', { name: 'Covered by package', exact: true }).waitFor();
  await capture('named-coverage', page.locator('#package-coverage'), true);
  await page.context().setStorageState('.local/ui-review/t05-desktop-value-customer.json');
  await page.goto('http://127.0.0.1:3000/learn/bookings/' + metadata.value.enrollmentId);
  await page.locator('#package-coverage').getByText('EGP 75.00', { exact: true }).waitFor();
  await capture('value-topup', page.locator('#package-coverage'), true);
  await page.context().setStorageState('.local/ui-review/t05-desktop-conversion-customer.json');
  await page.goto('http://127.0.0.1:3000/learn/packages?learner=' + metadata.conversion.learnerId);
  const conversion = page.locator('[data-proposal-id="' + metadata.conversion.proposalId + '"]');
  await conversion.locator('summary').click();
  await conversion.getByText('-3.6 seconds', { exact: true }).waitFor();
  await conversion.getByRole('heading', { name: 'Package terms', exact: true }).waitFor();
  await capture('conversion-terms', conversion, true);
  await capture('conversion-duration', conversion.getByText('Service-duration change', { exact: true }), true);
  await page.context().setStorageState('.local/ui-review/t05-desktop-admin.json');
  await page.goto('http://127.0.0.1:3000/admin/packages?package=' + metadata.purchase.packageId);
  const grant = page.getByRole('region', { name: 'Add approved bonus credits', exact: true });
  await grant.getByLabel('Approved bonus policy', { exact: true }).waitFor();
  await grant.getByText('Loading package records…', { exact: true }).waitFor({ state: 'hidden' });
  for (let pageNumber = 0; pageNumber < 20 && !await grant.getByLabel('Approved bonus policy', { exact: true }).locator('option').filter({ hasText: /subsidy/i }).count(); pageNumber += 1) {
    const more = grant.getByRole('button', { name: 'Load more records', exact: true });
    if (!await more.isVisible()) break;
    const response = page.waitForResponse(response => new URL(response.url()).pathname === '/api/v1/package-grant-policies' && response.status() === 200);
    await more.click(); await response;
    await grant.getByText('Loading more…', { exact: true }).waitFor({ state: 'hidden' });
  }
  await grant.getByLabel('Approved bonus policy', { exact: true }).locator('option').filter({ hasText: /subsidy/i }).first().waitFor({ state: 'attached' });
  const approvedPolicy = await grant.getByLabel('Approved bonus policy', { exact: true }).locator('option').evaluateAll(options => options.find(option => /subsidy/i.test(option.textContent))?.value);
  if (!approvedPolicy) throw new Error('Synthetic approved subsidy policy not found');
  await grant.getByLabel('Approved bonus policy', { exact: true }).selectOption(approvedPolicy);
  await grant.getByLabel('Credits', { exact: true }).fill('1');
  await grant.getByLabel('Allocated grant value (EGP)', { exact: true }).fill('500');
  await capture('admin-grant', grant);
  const reconciliation = page.locator('details').filter({ has: page.locator('summary').filter({ hasText: /^Quantity and value reconciliation$/ }) });
  await reconciliation.locator('summary').click();
  await reconciliation.getByText('Posted movements match the saved credit and value balances.', { exact: true }).waitFor();
  await capture('admin-reconciliation', reconciliation);
  return findings;
}
