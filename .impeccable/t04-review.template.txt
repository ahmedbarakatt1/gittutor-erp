async (page) => {
  const metadata = __METADATA__;
  const findings = [];
  let consoleErrors = [], warnings = [], failedResponses = [];
  page.on('console', message => { if (message.type() === 'error') consoleErrors.push(message.text().replace(/https?:\/\/\S+/g, '[url]')); if (message.type() === 'warning') warnings.push(message.text().replace(/https?:\/\/\S+/g, '[url]')); });
  page.on('pageerror', error => consoleErrors.push(error.message.replace(/https?:\/\/\S+/g, '[url]')));
  page.on('response', response => { if (response.status() >= 400) failedResponses.push({ path: new URL(response.url()).pathname, status: response.status() }); });
  async function capture(label, target, rtl = false) {
    const measurements = [];
    for (const viewport of [{ label: 'desktop', width: 1440, height: 1000 }, { label: 'mobile', width: 360, height: 780 }]) {
      await page.evaluate(() => document.activeElement?.blur());
      await page.setViewportSize({ width: viewport.width, height: viewport.height });
      await page.evaluate(() => document.fonts.ready);
      if (target) await target.evaluate(element => element.scrollIntoView({ block: 'start', behavior: 'instant' }));
      else await page.evaluate(() => scrollTo({ top: 0, behavior: 'instant' }));
      await page.screenshot({ path: '.impeccable/review/t04-' + label + '-' + viewport.label + '.png', caret: 'initial' });
      measurements.push({ viewport: viewport.width, overflow: await page.evaluate(() => document.documentElement.scrollWidth > innerWidth) });
    }
    await page.evaluate(() => { document.activeElement?.blur(); scrollTo({ top: 0, behavior: 'instant' }); });
    await page.keyboard.press('Tab');
    const focus = await page.evaluate(() => { const element = document.activeElement; const style = getComputedStyle(element); return { tag: element.tagName, text: element.textContent.trim().slice(0, 80), outlineStyle: style.outlineStyle, outlineWidth: style.outlineWidth }; });
    if (rtl) {
      await page.evaluate(() => { document.documentElement.dir = 'rtl'; });
      measurements.push({ viewport: 360, direction: 'rtl', overflow: await page.evaluate(() => document.documentElement.scrollWidth > innerWidth) });
      if (target) await target.evaluate(element => element.scrollIntoView({ block: 'start', behavior: 'instant' }));
      await page.screenshot({ path: '.impeccable/review/t04-' + label + '-rtl.png', caret: 'initial' });
      await page.evaluate(() => { document.documentElement.dir = 'ltr'; });
    }
    findings.push({ surface: label, measurements, focus, consoleErrors, warnings, failedResponses });
    consoleErrors = []; warnings = []; failedResponses = [];
  }
  await page.context().setStorageState('.local/ui-review/t04-desktop-guardian.json');
  await page.goto('http://127.0.0.1:3000/learn/bookings/' + metadata.booking.enrollmentIds[0]);
  await page.getByRole('heading', { name: 'Your exact quote', exact: true }).waitFor();
  await capture('customer-reservation', null, true);
  await capture('customer-quote', page.getByRole('region', { name: 'Your exact quote', exact: true }), true);
  await page.context().setStorageState('.local/ui-review/t04-desktop-proposal-added.json');
  await page.goto('http://127.0.0.1:3000/learn/bookings/' + metadata.amendment.provisionalEnrollmentId);
  await page.getByRole('heading', { name: 'Proposed quote', exact: true }).waitFor();
  await page.getByText('Review proposed change', { exact: true }).click();
  await page.getByText('Proposed tuition', { exact: true }).waitFor();
  await capture('provisional-seat', page.getByRole('region', { name: 'Proposed quote', exact: true }), true);
  const proposal = page.locator('details').filter({ has: page.getByText('Review proposed change', { exact: true }) }).first();
  await capture('provisional-terms', proposal, true);
  await page.context().setStorageState('.local/ui-review/t04-desktop-admin.json');
  await page.goto('http://127.0.0.1:3000/admin/schedule?occurrence=' + metadata.booking.occurrenceId);
  const privateHeading = page.getByRole('heading', { name: 'Confidential teaching terms', exact: true });
  await privateHeading.waitFor();
  await page.getByRole('button', { name: 'Approve exact tutor fee', exact: true }).waitFor({ state: 'visible' });
  await capture('admin-schedule', page.locator('#sessions'));
  await capture('admin-private', privateHeading);
  await page.context().setStorageState('.local/ui-review/t04-mobile-tutor.json');
  await page.goto('http://127.0.0.1:3000/tutor/schedule?occurrence=' + metadata.tutor.occurrenceId);
  const availability = page.getByRole('form', { name: 'Save availability', exact: true });
  await availability.getByRole('button', { name: 'Add a date exception', exact: true }).click();
  await availability.getByLabel('Date 1', { exact: true }).fill(metadata.tutor.localDate);
  await availability.getByLabel('Reason for this request', { exact: true }).fill('Synthetic rendered conflict review');
  const conflict = page.waitForResponse(response => new URL(response.url()).pathname === '/api/v1/availability' && response.request().method() === 'POST');
  await availability.getByRole('button', { name: 'Save availability', exact: true }).click();
  if ((await conflict).status() !== 409) throw new Error('Expected preserved availability conflict');
  const affected = page.getByRole('region', { name: 'Sessions affected by this change', exact: true });
  await affected.getByRole('link', { name: 'Review this session', exact: true }).waitFor();
  await capture('tutor-availability', page.locator('#availability'));
  await capture('tutor-conflict', affected);
  return findings;
}
