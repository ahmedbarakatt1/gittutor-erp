async (page) => {
  const metadata = __TRANSFER_METADATA__;
  const findings = [];
  let errors = 0, warnings = 0, failures = [];
  page.on('console', message => { if (message.type() === 'error') errors++; if (message.type() === 'warning') warnings++; });
  page.on('pageerror', () => errors++);
  page.on('response', response => { if (response.status() >= 400) failures.push({ path: new URL(response.url()).pathname, status: response.status() }); });
  async function capture(label, target, rtl) {
    const measurements = [];
    for (const viewport of [{ label: 'desktop', width: 1440, height: 1000 }, { label: 'mobile', width: 360, height: 780 }]) {
      await page.setViewportSize({ width: viewport.width, height: viewport.height });
      await page.evaluate(() => document.fonts.ready);
      await target.evaluate(element => element.scrollIntoView({ block: 'start', behavior: 'instant' }));
      await page.screenshot({ path: '.impeccable/review/t06-transfer-' + label + '-' + viewport.label + '.png' });
      measurements.push({ width: viewport.width, overflow: await page.evaluate(() => document.documentElement.scrollWidth > innerWidth) });
    }
    await page.getByRole('link', { name: 'Back to Git Tutor', exact: true }).focus();
    await page.keyboard.press('Tab');
    const focus = await page.evaluate(() => { const el = document.activeElement, style = getComputedStyle(el); return { tag: el.tagName, outlineWidth: style.outlineWidth, outlineStyle: style.outlineStyle }; });
    if (rtl) {
      await page.evaluate(() => { document.documentElement.dir = 'rtl'; });
      await target.evaluate(element => element.scrollIntoView({ block: 'start', behavior: 'instant' }));
      await page.screenshot({ path: '.impeccable/review/t06-transfer-' + label + '-rtl.png' });
      measurements.push({ width: 360, direction: 'rtl', overflow: await page.evaluate(() => document.documentElement.scrollWidth > innerWidth) });
      await page.evaluate(() => { document.documentElement.dir = 'ltr'; });
    }
    findings.push({ label, measurements, focus, consoleErrors: errors, consoleWarnings: warnings, failedResponses: failures });
    errors = 0; warnings = 0; failures = [];
  }
  await page.context().setStorageState('.local/ui-review/t06-transfer-desktop-customer.json');
  await page.goto('http://127.0.0.1:3000/learn/bookings/' + metadata.destinationEnrollmentId);
  const history = page.locator('[data-proposal-id="' + metadata.proposalId + '"]');
  await history.locator(':scope > summary').click();
  await history.getByText('The session transfer has been applied.', { exact: true }).waitFor();
  await capture('completed-booking', history, true);
  await page.context().setStorageState('.local/ui-review/t06-transfer-desktop-payer.json');
  await page.goto('http://127.0.0.1:3000/learn/payments/enrollment-transfer-consents/' + metadata.consentId);
  await page.getByText('Funds carried over', { exact: true }).waitFor();
  await capture('payer-history', page.getByRole('heading', { level: 1 }), true);
  return findings;
}
